\section{ Data Selection }

1. 
\begin{itemize}
    \item Student Performance Dataset (student-por.csv, Portuguese language course)
    \item https://archive.ics.uci.edu/dataset/320/student+performance
\end{itemize}

2. 
\begin{itemize}
    \item Regulated Domain: Education
\end{itemize}

3. 
\begin{itemize}
    \item Observations (Rows): 649 observation
\end{itemize}

4. 
\begin{itemize}
    \item Variables (Columns): 33 total variables
\end{itemize}

5. 
\begin{itemize}
    \item G3\_pass: A binary variable where 1 = Pass ( G3 >= 10) and 0 = Fail ( G3 < 10 ). This measures final course grade.
    \item G2\_pass: A binary variable where 1 = Pass ( G2 >= 10) and 0 = Fail ( G2 < 10 ). This measures second period grade.
\end{itemize}

6. 
\begin{itemize}
    \item 2 variables: Sex, Age
\end{itemize}

7.
\begin{itemize}
    \item Sex: Title IX of the Education Amendments of 1972
    \item Age: The Age Discrimination Act of 1975
\end{itemize}
